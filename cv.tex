\section{Selected Research Projects}
% BEGIN GENERATED PROJECTS
\begin{minipage}{\linewidth}
\textbf{Legal RAG: Retrieval Cues and Graph Structure}
\begin{itemize}
  \item Co-developed a Legal RAG pipeline combining LLM-based information extraction, Neo4j, dense/BM25 retrieval, reranking and controlled tests of retrieval cues versus graph structure.
  \item Contributed reproducible retrieval and evaluation infrastructure covering a 772,001-passage legal corpus and 4,584 HousingQA questions across 47 jurisdictions.
\end{itemize}
\end{minipage}\par\vspace{6pt}

\begin{minipage}{\linewidth}
\textbf{MJL-NER: Legal Entity Supervision in LLMs}
\begin{itemize}
  \item Co-developed controlled experiments on legal-entity supervision using QLoRA fine-tuning, vLLM inference, token-matched training and legal-citation counterfactuals.
  \item Contributed to evaluation across CaseHOLD and MultiEURLEX, including two 7B model families, three seeds and a reconstructed 197-document legal-entity resource.
\end{itemize}
\end{minipage}\par\vspace{6pt}

\begin{minipage}{\linewidth}
\textbf{Quantisation Limits of LLMs for Legal NLP}
\begin{itemize}
  \item Built a configurable evaluation framework for six open LLM configurations from 7B to 72B parameters, comparing post-training quantisation from FP8 to sub-2-bit variants.
  \item Executed and analysed 159 full-test result cells across LEDGAR, CaseHOLD and EURLEX, with paired statistical testing, structured-output audits, checkpoint-footprint measurement and backend controls.
\end{itemize}
\end{minipage}\par\vspace{6pt}

\begin{minipage}{\linewidth}
\textbf{LLM-Assisted Swiss Consultation Drafting}
\begin{itemize}
  \item Developed an LLM-assisted consultation-drafting backend with FastAPI, PostgreSQL and SSE streaming, including participant-linked transcripts and usage telemetry.
  \item Engineered transactional study-budget controls and recovery-aware multi-VM accounting, alongside representative multi-turn cost benchmarks and streaming load tests.
\end{itemize}
\end{minipage}\par\vspace{6pt}

\begin{minipage}{\linewidth}
\textbf{AI-Guided Online Deliberation}
\begin{itemize}
  \item Built a reproducible analysis pipeline for a four-arm randomised online-deliberation experiment covering 336 participants, 8,118 comments and 411,515 cleaned behavioural events.
  \item Implemented argument-tree compliance analysis, pre-registered statistical testing and LLM-based discourse annotation with blinded human validation and H100/Slurm inference.
\end{itemize}
\end{minipage}\par\vspace{6pt}

\begin{minipage}{\linewidth}
\textbf{LLM Code-Style Assessment and Human Evaluation}
\begin{itemize}
  \item Developed a Python code-style assessment pipeline integrating assignment context, Pylint and schema-validated feedback across OpenAI, Anthropic and local vLLM workflows.
  \item Built a shared 26-criterion human--AI annotation framework with source-integrity checks, reviewer adjudication and exact/overlap-aware evaluation of line-level findings.
\end{itemize}
\end{minipage}\par\vspace{6pt}
